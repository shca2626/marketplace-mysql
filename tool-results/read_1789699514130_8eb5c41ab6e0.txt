     1→'use client'
     2→
     3→import { useEffect, useMemo, useState } from 'react'
     4→import { motion } from 'framer-motion'
     5→import {
     6→  BadgeCheck,
     7→  ChevronRight,
     8→  Heart,
     9→  RotateCcw,
    10→  ShieldCheck,
    11→  ShoppingCart,
    12→  Star,
    13→  Truck,
    14→  X,
    15→  Zap,
    16→  ZoomIn,
    17→} from 'lucide-react'
    18→import { toast } from '@/hooks/use-toast'
    19→import { useMarketplace } from '@/store/marketplace'
    20→import { useApi } from '@/hooks/use-api'
    21→import type { ProductDTO, ReviewDTO } from '@/lib/types'
    22→import { discountPercent, formatDate, formatNumber, formatUSD, initials, timeAgo } from '@/lib/format'
    23→import { StarPicker } from './star-rating'
    24→import { QuantityInput } from './quantity-input'
    25→import { ProductCard } from './product-card'
    26→import { Button } from '@/components/ui/button'
    27→import { Badge } from '@/components/ui/badge'
    28→import { Card } from '@/components/ui/card'
    29→import { Skeleton } from '@/components/ui/skeleton'
    30→import { Avatar, AvatarFallback } from '@/components/ui/avatar'
    31→import { Separator } from '@/components/ui/separator'
    32→import { Tabs, TabsContent, TabsList, TabsTrigger } from '@/components/ui/tabs'
    33→import { Textarea } from '@/components/ui/textarea'
    34→import { Dialog, DialogContent, DialogTitle } from '@/components/ui/dialog'
    35→import { cn } from '@/lib/utils'
    36→
    37→interface ProductDetailResponse {
    38→  product: ProductDTO
    39→  reviews: ReviewDTO[]
    40→  seller: {
    41→    id: string
    42→    name: string
    43→    storeName: string | null
    44→    storeBio: string | null
    45→    memberSince: string
    46→    activeListings: number
    47→  } | null
    48→  related: ProductDTO[]
    49→}
    50→
    51→export function ProductView() {
    52→  const productId = useMarketplace((s) => s.params.productId)
    53→  const navigate = useMarketplace((s) => s.navigate)
    54→  const addToCart = useMarketplace((s) => s.addToCart)
    55→  const setCartOpen = useMarketplace((s) => s.setCartOpen)
    56→  const user = useMarketplace((s) => s.user)
    57→  const openAuth = useMarketplace((s) => s.openAuth)
    58→  const trackView = useMarketplace((s) => s.trackView)
    59→
    60→  const { data, loading, setData } = useApi<ProductDetailResponse>(
    61→    productId ? `/api/products/${productId}` : null
    62→  )
    63→
    64→  const [activeImage, setActiveImage] = useState(0)
    65→  const [lightboxOpen, setLightboxOpen] = useState(false)
    66→  const [quantity, setQuantity] = useState(1)
    67→  const [reviewRating, setReviewRating] = useState(5)
    68→  const [reviewComment, setReviewComment] = useState('')
    69→  const [submittingReview, setSubmittingReview] = useState(false)
    70→
    71→  const product = data?.product
    72→
    73→  // Record the visit for the "Recently viewed" row on home
    74→  useEffect(() => {
    75→    if (productId) trackView(productId)
    76→  }, [productId, trackView])
    77→
    78→  const discount = useMemo(
    79→    () => (product ? discountPercent(product.price, product.compareAtPrice) : null),
    80→    [product]
    81→  )
    82→
    83→  const handleAddToCart = (openDrawer = false) => {
    84→    if (!product) return
    85→    const res = addToCart(product, quantity)
    86→    if (res.ok) {
    87→      toast({ title: 'Added to cart', description: `${quantity} × ${product.title}` })
    88→      if (openDrawer) setCartOpen(true)
    89→    } else {
    90→      toast({ title: 'Cannot add to cart', description: res.message, variant: 'destructive' })
    91→    }
    92→  }
    93→
    94→  const handleBuyNow = () => {
    95→    if (!product) return
    96→    const res = addToCart(product, quantity)
    97→    if (res.ok) navigate('checkout')
    98→    else toast({ title: 'Cannot add to cart', description: res.message, variant: 'destructive' })
    99→  }
   100→
   101→  const handleFavorite = async () => {
   102→    if (!product) return
   103→    if (!user) {
   104→      openAuth('login', { view: 'product', productId: product.id })
   105→      return
   106→    }
   107→    try {
   108→      const res = await fetch('/api/favorites', {
   109→        method: 'POST',
   110→        headers: { 'Content-Type': 'application/json' },
   111→        body: JSON.stringify({ productId: product.id }),
   112→      })
   113→      const json = await res.json()
   114→      if (!res.ok) {
   115→        toast({ title: 'Error', description: json.error, variant: 'destructive' })
   116→        return
   117→      }
   118→      setData(
   119→        data ? { ...data, product: { ...product, favorited: json.favorited } } : data,
   120→      )
   121→      toast({
   122→        title: json.favorited ? 'Saved to favorites' : 'Removed from favorites',
   123→        description: product.title,
   124→      })
   125→      window.dispatchEvent(new CustomEvent('favorites-changed'))
   126→    } catch {
   127→      toast({ title: 'Error', description: 'Network error', variant: 'destructive' })
   128→    }
   129→  }
   130→
   131→  const submitReview = async (e: React.FormEvent) => {
   132→    e.preventDefault()
   133→    if (!product) return
   134→    if (!user) {
   135→      openAuth('login', { view: 'product', productId: product.id })
   136→      return
   137→    }
   138→    if (reviewComment.trim().length < 3) {
   139→      toast({ title: 'Review too short', description: 'Please write at least a few words', variant: 'destructive' })
   140→      return
   141→    }
   142→    setSubmittingReview(true)
   143→    try {
   144→      const res = await fetch('/api/reviews', {
   145→        method: 'POST',
   146→        headers: { 'Content-Type': 'application/json' },
   147→        body: JSON.stringify({ productId: product.id, rating: reviewRating, comment: reviewComment.trim() }),
   148→      })
   149→      const json = await res.json()
   150→      if (!res.ok) {
   151→        toast({ title: 'Error', description: json.error, variant: 'destructive' })
   152→        return
   153→      }
   154→      // Update local state: refresh from server
   155→      const fresh = await fetch(`/api/products/${product.id}`, { cache: 'no-store' })
   156→      if (fresh.ok) setData(await fresh.json())
   157→      setReviewComment('')
   158→      setReviewRating(5)
   159→      toast({ title: 'Review submitted', description: 'Thanks for sharing your thoughts!' })
   160→    } catch {
   161→      toast({ title: 'Error', description: 'Network error', variant: 'destructive' })
   162→    } finally {
   163→      setSubmittingReview(false)
   164→    }
   165→  }
   166→
   167→  if (loading || !product) {
   168→    return (
   169→      <div className="mx-auto max-w-7xl px-4 py-8">
   170→        <div className="grid gap-8 lg:grid-cols-2">
   171→          <Skeleton className="aspect-square w-full rounded-2xl" />
   172→          <div className="space-y-4">
   173→            <Skeleton className="h-6 w-24" />
   174→            <Skeleton className="h-10 w-3/4" />
   175→            <Skeleton className="h-5 w-1/2" />
   176→            <Skeleton className="h-8 w-32" />
   177→            <Skeleton className="h-24 w-full" />
   178→            <Skeleton className="h-12 w-full" />
   179→          </div>
   180→        </div>
   181→      </div>
   182→    )
   183→  }
   184→
   185→  const ratingBuckets = [5, 4, 3, 2, 1].map((star) => ({
   186→    star,
   187→    count: data.reviews.filter((r) => r.rating === star).length,
   188→  }))
   189→  const totalReviews = data.reviews.length
   190→  const myReview = user ? data.reviews.find((r) => r.user.id === user.id) : undefined
   191→
   192→  return (
   193→    <div className="mx-auto max-w-7xl px-4 py-8">
   194→      {/* Breadcrumb */}
   195→      <nav className="mb-6 flex flex-wrap items-center gap-1 text-sm text-muted-foreground" aria-label="Breadcrumb">
   196→        <button className="hover:text-primary" onClick={() => navigate('home')}>
   197→          Home
   198→        </button>
   199→        <ChevronRight size={13} />
   200→        <button className="hover:text-primary" onClick={() => navigate('catalog', { category: product.category.slug })}>
   201→          {product.category.name}
   202→        </button>
   203→        <ChevronRight size={13} />
   204→        <span className="line-clamp-1 max-w-64 font-medium text-foreground">{product.title}</span>
   205→      </nav>
   206→
   207→      <div className="grid gap-10 lg:grid-cols-2">
   208→        {/* ── Gallery ── */}
   209→        <div>
   210→          <motion.div
   211→            key={activeImage}
   212→            initial={{ opacity: 0.4, scale: 0.99 }}
   213→            animate={{ opacity: 1, scale: 1 }}
   214→            transition={{ duration: 0.25 }}
   215→            className="group relative overflow-hidden rounded-2xl border bg-muted"
   216→          >
   217→            {product.images[activeImage] ? (
   218→              <button
   219→                className="block w-full cursor-zoom-in"
   220→                onClick={() => setLightboxOpen(true)}
   221→                aria-label="Enlarge image"
   222→              >
   223→                <img
   224→                  src={product.images[activeImage]}
   225→                  alt={`${product.title} — image ${activeImage + 1}`}
   226→                  className="aspect-square w-full object-cover transition-transform duration-500 group-hover:scale-[1.03]"
   227→                />
   228→                <span className="absolute right-3 top-3 flex h-9 w-9 items-center justify-center rounded-full bg-background/80 text-foreground opacity-0 shadow-sm backdrop-blur transition-opacity duration-200 group-hover:opacity-100">
   229→                  <ZoomIn size={16} />
   230→                </span>
   231→              </button>
   232→            ) : (
   233→              <div className="flex aspect-square items-center justify-center text-muted-foreground">No image</div>
   234→            )}
   235→            {discount !== null && (
   236→              <Badge className="absolute left-3 top-3 bg-red-500/95 text-sm font-bold text-white hover:bg-red-500">
   237→                -{discount}%
   238→              </Badge>
   239→            )}
   240→          </motion.div>
   241→          {product.images.length > 1 && (
   242→            <div className="mt-3 flex gap-2.5">
   243→              {product.images.map((img, i) => (
   244→                <button
   245→                  key={i}
   246→                  onClick={() => setActiveImage(i)}
   247→                  className={cn(
   248→                    'h-20 w-20 shrink-0 overflow-hidden rounded-xl border-2 transition-all',
   249→                    i === activeImage ? 'border-primary' : 'border-transparent opacity-60 hover:opacity-100'
   250→                  )}
   251→                  aria-label={`Show image ${i + 1}`}
   252→                >
   253→                  <img src={img} alt="" className="h-full w-full object-cover" />
   254→                </button>
   255→              ))}
   256→            </div>
   257→          )}
   258→        </div>
   259→
   260→        {/* ── Info ── */}
   261→        <motion.div initial={{ opacity: 0, y: 12 }} animate={{ opacity: 1, y: 0 }} transition={{ duration: 0.3 }}>
   262→          <Badge variant="secondary" className="rounded-full text-xs font-medium">
   263→            {product.category.name}
   264→          </Badge>
   265→          <h1 className="mt-3 text-2xl font-bold leading-snug tracking-tight sm:text-3xl">{product.title}</h1>
   266→
   267→          <div className="mt-3 flex flex-wrap items-center gap-x-4 gap-y-1 text-sm">
   268→            <div className="flex items-center gap-1.5">
   269→              <Star size={15} className="fill-amber-400 text-amber-400" />
   270→              <span className="font-semibold">{product.rating.toFixed(1)}</span>
   271→              <button
   272→                className="text-muted-foreground underline-offset-2 hover:underline"
   273→                onClick={() => document.getElementById('reviews-tab')?.click()}
   274→              >
   275→                {product.reviewCount} reviews
   276→              </button>
   277→            </div>
   278→            <span className="text-muted-foreground">·</span>
   279→            <span className="text-muted-foreground">{formatNumber(product.soldCount)} sold</span>
   280→          </div>
   281→
   282→          <div className="mt-5 flex items-end gap-3">
   283→            <span className="text-4xl font-extrabold tracking-tight">{formatUSD(product.price)}</span>
   284→            {product.compareAtPrice && (
   285→              <span className="pb-1 text-lg text-muted-foreground line-through">
   286→                {formatUSD(product.compareAtPrice)}
   287→              </span>
   288→            )}
   289→            {discount !== null && (
   290→              <Badge variant="destructive" className="mb-1.5">
   291→                Save {discount}%
   292→              </Badge>
   293→            )}
   294→          </div>
   295→
   296→          <p className="mt-4 leading-relaxed text-muted-foreground">{product.description}</p>
   297→
   298→          {/* Stock */}
   299→          <div className="mt-5 flex items-center gap-2 text-sm">
   300→            <span
   301→              className={cn(
   302→                'h-2 w-2 rounded-full',
   303→                product.stock === 0 ? 'bg-destructive' : product.stock <= 5 ? 'bg-amber-500' : 'bg-emerald-500'
   304→              )}
   305→            />
   306→            {product.stock === 0 ? (
   307→              <span className="font-medium text-destructive">Out of stock</span>
   308→            ) : product.stock <= 5 ? (
   309→              <span className="font-medium text-amber-600 dark:text-amber-400">
   310→                Selling fast — only {product.stock} left
   311→              </span>
   312→            ) : (
   313→              <span className="text-muted-foreground">{product.stock} in stock</span>
   314→            )}
   315→          </div>
   316→
   317→          {/* Quantity + actions */}
   318→          <div className="mt-6 flex flex-wrap items-center gap-3">
   319→            <QuantityInput value={quantity} onChange={setQuantity} max={Math.max(1, product.stock)} />
   320→            <Button
   321→              size="lg"
   322→              className="h-12 flex-1 rounded-xl text-base sm:flex-none sm:px-8"
   323→              disabled={product.stock === 0}
   324→              onClick={() => handleAddToCart(false)}
   325→            >
   326→              <ShoppingCart size={17} className="mr-2" /> Add to cart
   327→            </Button>
   328→            <Button
   329→              size="lg"
   330→              variant="secondary"
   331→              className="h-12 rounded-xl border text-base sm:px-8"
   332→              disabled={product.stock === 0}
   333→              onClick={handleBuyNow}
   334→            >
   335→              <Zap size={16} className="mr-2" /> Buy now
   336→            </Button>
   337→            <Button
   338→              size="lg"
   339→              variant="outline"
   340→              className="h-12 w-12 rounded-xl p-0"
   341→              onClick={handleFavorite}
   342→              aria-label="Toggle favorite"
   343→            >
   344→              <Heart
   345→                size={19}
   346→                className={cn('transition-colors', product.favorited && 'fill-red-500 text-red-500')}
   347→              />
   348→            </Button>
   349→          </div>
   350→
   351→          {/* Perks */}
   352→          <div className="mt-7 grid grid-cols-3 gap-3">
   353→            {[
   354→              { icon: Truck, title: 'Fast shipping', text: 'Free over $75' },
   355→              { icon: RotateCcw, title: 'Easy returns', text: 'Within 30 days' },
   356→              { icon: ShieldCheck, title: 'Warranty', text: '1-year coverage' },
   357→            ].map((perk) => (
   358→              <div key={perk.title} className="rounded-xl border bg-card p-3.5 text-center">
   359→                <perk.icon size={18} className="mx-auto text-primary" />
   360→                <p className="mt-2 text-xs font-semibold leading-tight">{perk.title}</p>
   361→                <p className="mt-0.5 text-[11px] text-muted-foreground">{perk.text}</p>
   362→              </div>
   363→            ))}
   364→          </div>
   365→
   366→          {/* Seller card — links to the public storefront */}
   367→          {data.seller && (
   368→            <Card
   369→              className="group mt-7 cursor-pointer p-5 transition-all hover:border-primary/40 hover:shadow-md"
   370→              onClick={() => navigate('store', { sellerId: data.seller!.id })}
   371→              role="link"
   372→              aria-label={`Visit ${data.seller.storeName ?? data.seller.name} store`}
   373→            >
   374→              <div className="flex items-center gap-4">
   375→                <Avatar className="h-13 w-13 border">
   376→                  <AvatarFallback className="bg-primary/10 font-bold text-primary">
   377→                    {initials(data.seller.storeName ?? data.seller.name)}
   378→                  </AvatarFallback>
   379→                </Avatar>
   380→                <div className="min-w-0 flex-1">
   381→                  <div className="flex items-center gap-1.5">
   382→                    <p className="truncate font-semibold group-hover:text-primary">
   383→                      {data.seller.storeName ?? data.seller.name}
   384→                    </p>
   385→                    <BadgeCheck size={15} className="shrink-0 text-primary" />
   386→                  </div>
   387→                  <p className="mt-0.5 text-xs text-muted-foreground">
   388→                    Member since {formatDate(data.seller.memberSince)} · {data.seller.activeListings} listings
   389→                  </p>
   390→                  {data.seller.storeBio && (
   391→                    <p className="mt-2 line-clamp-2 text-xs leading-relaxed text-muted-foreground">
   392→                      {data.seller.storeBio}
   393→                    </p>
   394→                  )}
   395→                </div>
   396→                <span className="hidden shrink-0 items-center gap-1 text-xs font-semibold text-primary sm:flex">
   397→                  View store
   398→                  <ChevronRight size={14} className="transition-transform group-hover:translate-x-0.5" />
   399→                </span>
   400→              </div>
   401→            </Card>
   402→          )}
   403→        </motion.div>
   404→      </div>
   405→
   406→      {/* ── Image lightbox ── */}
   407→      <Dialog open={lightboxOpen} onOpenChange={setLightboxOpen}>
   408→        <DialogContent
   409→          className="max-w-4xl overflow-hidden rounded-2xl border-none bg-background/95 p-0 shadow-2xl backdrop-blur sm:rounded-3xl [&>button]:hidden"
   410→          onKeyDown={(e) => {
   411→            if (e.key === 'ArrowRight' && activeImage < product.images.length - 1) setActiveImage(activeImage + 1)
   412→            if (e.key === 'ArrowLeft' && activeImage > 0) setActiveImage(activeImage - 1)
   413→          }}
   414→        >
   415→          <DialogTitle className="sr-only">{product.title} — image gallery</DialogTitle>
   416→          <div className="relative">
   417→            <img
   418→              src={product.images[activeImage]}
   419→              alt={`${product.title} — image ${activeImage + 1} enlarged`}
   420→              className="max-h-[75vh] w-full object-contain"
   421→            />
   422→            <Button
   423→              variant="secondary"
   424→              size="icon"
   425→              className="absolute right-3 top-3 h-10 w-10 rounded-full bg-background/80 shadow-md backdrop-blur"
   426→              onClick={() => setLightboxOpen(false)}
   427→              aria-label="Close image viewer"
   428→            >
   429→              <X size={18} />
   430→            </Button>
   431→            {product.images.length > 1 && (
   432→              <>
   433→                <Button
   434→                  variant="secondary"
   435→                  size="icon"
   436→                  className="absolute left-3 top-1/2 h-11 w-11 -translate-y-1/2 rounded-full bg-background/80 shadow-md backdrop-blur transition-transform hover:scale-105 disabled:opacity-30"
   437→                  disabled={activeImage === 0}
   438→                  onClick={() => setActiveImage((i) => Math.max(0, i - 1))}
   439→                  aria-label="Previous image"
   440→                >
   441→                  <ChevronRight size={20} className="rotate-180" />
   442→                </Button>
   443→                <Button
   444→                  variant="secondary"
   445→                  size="icon"
   446→                  className="absolute right-3 top-1/2 h-11 w-11 -translate-y-1/2 rounded-full bg-background/80 shadow-md backdrop-blur transition-transform hover:scale-105 disabled:opacity-30"
   447→                  disabled={activeImage === product.images.length - 1}
   448→                  onClick={() => setActiveImage((i) => Math.min(product.images.length - 1, i + 1))}
   449→                  aria-label="Next image"
   450→                >
   451→                  <ChevronRight size={20} />
   452→                </Button>
   453→                <div className="absolute bottom-3 left-1/2 flex -translate-x-1/2 gap-1.5 rounded-full bg-background/80 px-3 py-1.5 shadow-sm backdrop-blur">
   454→                  {product.images.map((img, i) => (
   455→                    <button
   456→                      key={i}
   457→                      onClick={() => setActiveImage(i)}
   458→                      aria-label={`Show image ${i + 1}`}
   459→                      className={cn(
   460→                        'h-2 rounded-full transition-all',
   461→                        i === activeImage ? 'w-6 bg-primary' : 'w-2 bg-muted-foreground/40 hover:bg-muted-foreground/70'
   462→                      )}
   463→                    />
   464→                  ))}
   465→                </div>
   466→              </>
   467→            )}
   468→          </div>
   469→        </DialogContent>
   470→      </Dialog>
   471→
   472→      {/* ── Tabs: description / reviews ── */}
   473→      <Tabs defaultValue="description" className="mt-14">
   474→        <TabsList className="h-11 w-full justify-start gap-1 overflow-x-auto rounded-xl bg-muted/60 p-1 sm:w-auto">
   475→          <TabsTrigger value="description" className="rounded-lg px-5 text-sm">
   476→            Description
   477→          </TabsTrigger>
   478→          <TabsTrigger value="reviews" id="reviews-tab" className="rounded-lg px-5 text-sm">
   479→            Reviews ({product.reviewCount})
   480→          </TabsTrigger>
   481→        </TabsList>
   482→
   483→        <TabsContent value="description" className="mt-6">
   484→          <Card className="p-6">
   485→            <h2 className="text-lg font-bold">About this item</h2>
   486→            <p className="mt-3 leading-relaxed text-muted-foreground">{product.description}</p>
   487→            <Separator className="my-5" />
   488→            <div className="grid gap-3 text-sm sm:grid-cols-2">
   489→              <div className="flex justify-between rounded-lg bg-muted/50 px-4 py-2.5">
   490→                <span className="text-muted-foreground">Category</span>
   491→                <span className="font-medium">{product.category.name}</span>
   492→              </div>
   493→              <div className="flex justify-between rounded-lg bg-muted/50 px-4 py-2.5">
   494→                <span className="text-muted-foreground">Listed</span>
   495→                <span className="font-medium">{formatDate(product.createdAt)}</span>
   496→              </div>
   497→              <div className="flex justify-between rounded-lg bg-muted/50 px-4 py-2.5">
   498→                <span className="text-muted-foreground">Units sold</span>
   499→                <span className="font-medium">{formatNumber(product.soldCount)}</span>
   500→              </div>
   501→              <div className="flex justify-between rounded-lg bg-muted/50 px-4 py-2.5">
   502→                <span className="text-muted-foreground">Ships from</span>
   503→                <span className="font-medium">Vendora fulfillment</span>
   504→              </div>
   505→            </div>
   506→          </Card>
   507→        </TabsContent>
