#!/bin/bash
# ─────────────────────────────────────────────────────────────────────────────
# start-mysql.sh — idempotent MariaDB restart helper (user-space install).
#
# Environment: no root, no systemd. Binary install at /home/z/mysql/usr,
# datadir /home/z/mysql-data, port 3306, socket /home/z/mysql/mysql.sock.
#
# IMPORTANT (sandbox quirk, verified 2026-09-20): background processes only
# survive across shell sessions when spawned through a double-fork subshell
# `( ... & )`. A plain `cmd &` child of the tool shell is reaped; a detached
# `( setsid cmd & )` orphan survives. This script uses that pattern.
#
# If the datadir is missing (fresh host), the database itself is gone: re-create
# with the recovery runbook in /home/z/my-project/whois.md → "Disaster recovery".
# ─────────────────────────────────────────────────────────────────────────────
set -u

MYSQL_HOME=/home/z/mysql
DATADIR=/home/z/mysql-data
SOCKET=$MYSQL_HOME/mysql.sock
PIDFILE=$MYSQL_HOME/mysqld.pid
BIN=$MYSQL_HOME/usr/bin/mariadbd
export LD_LIBRARY_PATH="$MYSQL_HOME/usr/lib:$MYSQL_HOME/usr/lib/x86_64-linux-gnu"

if [ ! -x "$BIN" ]; then
  echo "[mysql] ERROR: $BIN not found — MariaDB user-space install is missing."
  echo "[mysql] Follow the disaster-recovery runbook (whois.md) to reinstall + reseed."
  exit 1
fi
if [ ! -d "$DATADIR" ] || [ -z "$(ls -A "$DATADIR" 2>/dev/null)" ]; then
  echo "[mysql] ERROR: datadir $DATADIR is missing or empty — data is gone."
  echo "[mysql] Follow the disaster-recovery runbook (whois.md) to re-init + reseed."
  exit 1
fi

# Already up?
if [ -S "$SOCKET" ] && "$MYSQL_HOME/usr/bin/mariadb-admin" --protocol=socket -S "$SOCKET" ping >/dev/null 2>&1; then
  echo "[mysql] already running (socket ping ok)"
  exit 0
fi

# Stale socket/pid from an unclean shutdown
rm -f "$SOCKET" "$PIDFILE"

echo "[mysql] starting mariadbd..."
(
  cd "$MYSQL_HOME/usr" || exit 1
  setsid "$BIN" --defaults-file="$MYSQL_HOME/my.cnf" < /dev/null >> "$MYSQL_HOME/mysqld.out" 2>&1 &
)

for i in $(seq 1 60); do
  if [ -S "$SOCKET" ] && "$MYSQL_HOME/usr/bin/mariadb-admin" --protocol=socket -S "$SOCKET" ping >/dev/null 2>&1; then
    echo "[mysql] ready on :3306 (socket $SOCKET)"
    exit 0
  fi
  sleep 1
done

echo "[mysql] ERROR: server did not become ready in 60s — last error log lines:"
tail -20 "$MYSQL_HOME/mysqld.err" 2>/dev/null || tail -20 "$MYSQL_HOME/mysqld.out" 2>/dev/null
exit 1
